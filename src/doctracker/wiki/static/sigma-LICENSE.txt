sigma.min.js bundles the libraries of DATABASE › Universe into one file, exposing the global
`SigmaBundle` (Graph, Sigma, EdgeCurveProgram, indexParallelEdgesIndex, forceAtlas2, noverlap).
Every one of them is under the MIT licence, reproduced below.

  sigma 3.0.2, graphology 0.26.0, graphology-utils 2.5.2, events 3.3.0,
  graphology-layout-forceatlas2 0.10.1, graphology-layout-noverlap 0.4.2, @sigma/edge-curve 3.1.0

Built with esbuild 0.25.0 (MIT) from this entry file:

  import Graph from "graphology";
  import Sigma from "sigma";
  import EdgeCurveProgram, { indexParallelEdgesIndex } from "@sigma/edge-curve";
  import forceAtlas2 from "graphology-layout-forceatlas2";
  import noverlap from "graphology-layout-noverlap";
  export { Graph, Sigma, EdgeCurveProgram, indexParallelEdgesIndex, forceAtlas2, noverlap };

  esbuild entry.js --bundle --minify --format=iife --global-name=SigmaBundle --legal-comments=none --outfile=sigma.min.js
